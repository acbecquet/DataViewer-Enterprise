Option Explicit

Private Sub Workbook_Open()
    ' Register sample-navigation hotkeys.
    '   + = Shift, ^ = Ctrl  ->  "+^." = Ctrl+Shift+. and "+^," = Ctrl+Shift+,
    On Error Resume Next
    Application.OnKey "+^.", "JumpRight12"
    Application.OnKey "+^,", "JumpLeft12"
End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)
    ' Restore Excel default behavior so the bindings don't leak into other
    ' workbooks the user opens in the same Excel session.
    On Error Resume Next
    Application.OnKey "+^."
    Application.OnKey "+^,"
End Sub
